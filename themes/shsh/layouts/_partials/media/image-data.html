{{- $media := partial "media/resolve.html" . -}} {{- $width := 0 -}}{{- $height := 0 -}} {{- $sources := slice -}} {{- $resource := $media.resource -}} {{- $processable := false -}} {{- $svg := "" -}} {{- if $resource -}} {{- if not (reflect.IsImageResource $resource) }} {{ errorf "shsh: expected image resource %q" .src }} {{ end -}} {{- $type := $resource.MediaType.SubType -}} {{- $isSVG := eq (lower (path.Ext $resource.Name)) ".svg" -}} {{- /* Hugo rereads Content on every access, so read it once and only for formats sniffed here. */ -}} {{- $bytes := "" -}} {{- if or $isSVG (in (slice "gif" "png" "webp" "avif") $type) }} {{ $bytes = $resource.Content }} {{ end -}} {{- if $isSVG }}{{ $svg = $bytes }}{{ end -}} {{- $animated := false -}} {{- if and (eq $type "gif") (ge (len $bytes) 13) }} {{ $animated = partial "media/gif-animated.html" (dict "bytes" $bytes) }} {{ end -}} {{- if eq $type "png" }} {{ $animated = partial "media/png-animated.html" (dict "bytes" $bytes "offset" 8) }} {{ end -}} {{- $webp := dict -}} {{- if eq $type "webp" -}} {{- $webp = partial "media/webp-info.html" (dict "bytes" $bytes "offset" 12) -}} {{- $animated = $webp.animated -}} {{- if $animated }} {{ $width = $webp.width }}{{ $height = $webp.height }} {{ end -}} {{- end -}} {{- if and (eq $type "avif") (findRE `avis` $bytes) }} {{ $animated = true }} {{ end -}} {{- if and (not $width) (reflect.IsImageResourceWithMeta $resource) -}} {{- $width = $resource.Width -}}{{- $height = $resource.Height -}} {{- end -}} {{- $processable = and (not $animated) (reflect.IsImageResourceProcessable $resource) -}} {{- if $processable -}} {{- /* Treat PNG, GIF and lossless WebP as screenshots and line art, the rest as photos. */ -}} {{- $screenshot := in (slice "png" "gif") $type -}} {{- if eq $type "webp" }}{{ $screenshot = $webp.lossless }}{{ end -}} {{- /* Hugo's lossless AVIF passes through BT.601 YUV and outgrows lossless WebP. */ -}} {{- /* Screenshots get near-lossless 4:4:4 AVIF instead; photos follow imaging.avif. */ -}} {{- /* AVIF comes first: browsers take the first typed source they support. */ -}} {{- $formats := slice (dict "type" "avif" "spec" (cond $screenshot "avif lossy catmullrom q85 drawing" "avif lossy")) (dict "type" "webp" "spec" (cond $screenshot "webp lossless catmullrom" "webp lossy")) -}} {{- range $format := $formats -}} {{- $candidates := slice -}} {{- range slice 360 720 1080 1440 -}} {{- if lt . $width }} {{ $candidates = $candidates | append ($resource.Process (printf "resize %dx %s" . $format.spec)) }} {{ end -}} {{- end -}} {{- /* Always include the native width so each source covers every display density. */ -}} {{- /* An original in the source's own format already belongs there at its own width. */ -}} {{- if eq $type $format.type -}} {{- $candidates = $candidates | append ($resource | fingerprint) -}} {{- else -}} {{- $candidates = $candidates | append ($resource.Process (printf "resize %dx %s" $width $format.spec)) -}} {{- end -}} {{- /* Covers repeat on list pages; the hashed src and format determine the candidates. */ -}} {{- $srcset := partialCached "media/srcset.html" $candidates $media.src $format.type -}} {{- $sources = $sources | append (dict "type" (printf "image/%s" $format.type) "srcset" $srcset) -}} {{- end -}} {{- end -}} {{- else if $media.static -}} {{- if eq (lower (path.Ext $media.static)) ".svg" }} {{ $svg = readFile $media.static }} {{ end -}} {{- if in (slice ".png" ".jpg" ".jpeg" ".gif" ".bmp" ".tif" ".tiff" ".webp" ".avif" ".heic" ".heif") (lower (path.Ext $media.static)) -}} {{- $config := images.Config $media.static -}} {{- $width = $config.Width -}}{{- $height = $config.Height -}} {{- end -}} {{- end -}} {{- with $svg -}} {{- $dimensions := partial "media/svg-dimensions.html" . -}} {{- $width = $dimensions.width | default 0 -}}{{- $height = $dimensions.height | default 0 -}} {{- end -}} {{- range $key := slice "width" "height" -}} {{- $value := index $ $key -}} {{- if and (ne $value nil) (ne $value "") -}} {{- if not (findRE `^[1-9][0-9]*$` (printf "%v" $value)) }} {{ errorf "shsh: image %s must be a positive integer" $key }} {{ end -}} {{- end -}} {{- end -}} {{- if not $width }}{{ $width = int (.width | default 0) }}{{ end -}} {{- if not $height }}{{ $height = int (.height | default 0) }}{{ end -}} {{- return (merge $media (dict "width" $width "height" $height "sources" $sources)) -}}
